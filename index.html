export function renderAgendaList() { const box = $('agList'); const count = $('agCount'); if (!box) return; box.replaceChildren(); const filterClass = $('agFilterClass').value; let items = (db.classAgenda || []).slice(); if (filterClass) items = items.filter(x => x.class === filterClass); // Trier : non-faits d'abord, puis par date croissante items.sort((a, b) => { if (a.done !== b.done) return a.done ? 1 : -1; return a.date.localeCompare(b.date); }); if (count) { const total = items.length; const done = items.filter(x => x.done).length; const pending = total - done; count.textContent = 'المواعيد: ' + total + ' (منجزة: ' + done + ' — قادمة: ' + pending + ')'; } if (items.length === 0) { box.append(el('div', { class: 'empty-list', textContent: 'لا توجد مواعيد.' })); return; } // On utilise le tableau stylé const table = el('table', { class: 'dail-table' }); const thead = el('thead'); thead.append(el('tr', {}, el('th', { textContent: 'التاريخ', class: 'col-n', style: 'width:120px' }), el('th', { textContent: 'الفصل', style: 'text-align:center;width:100px' }), el('th', { textContent: 'العنوان', class: 'col-name' }), el('th', { textContent: 'النوع', style: 'text-align:center;width:100px' }), el('th', { textContent: 'الحالة', style: 'text-align:center;width:120px' }), el('th', { textContent: 'إجراء', class: 'col-del' }) )); table.append(thead); const tbody = el('tbody'); items.forEach(item => { const status = getStatus(item); const statusLabels = { done: '✅ منجز', overdue: '⚠️ متأخر', soon: '🔔 قريب', normal: '—', }; const statusClasses = { done: 'done', overdue: 'overdue', soon: 'soon', normal: 'normal', }; let rowClass = ''; if (status === 'soon') rowClass = 'ag-soon'; else if (status === 'overdue') rowClass = 'ag-overdue'; else if (status === 'done') rowClass = 'ag-done'; const titleCell = el('td', {}, el('div', { style: 'font-weight:700;font-size:16px;color:#0f172a', textContent: item.title }), item.desc ? el('div', { style: 'font-size:13px;color:#64748b;margin-top:2px', textContent: item.desc }) : null ); const btnDone = el('button', { class: 'btn-icon-sm', textContent: item.done ? '↩️' : '✓', title: item.done ? 'إعادة فتح' : 'تم', style: 'background:#dcf2e5;color:#1f7a4d;margin-inline-end:4px', onclick: () => toggleDone(item.id) }); const btnDel = el('button', { class: 'btn-icon-sm', textContent: '🗑️', title: 'حذف', style: 'background:#fee2e2;color:#b3261e', onclick: () => deleteAgendaItem(item.id) }); const tr = el('tr', { class: rowClass }, el('td', { textContent: item.date, style: 'direction:ltr;font-size:14px' }), el('td', { textContent: item.class, style: 'text-align:center' }), titleCell, el('td', {}, el('span', { class: 'ag-type-badge ' + item.type, textContent: TYPE_LABELS[item.type] }) ), el('td', {}, el('span', { class: 'ag-status ' + statusClasses[status], textContent: statusLabels[status] }) ), el('td', {}, btnDone, btnDel) ); tbody.append(tr); }); table.append(tbody); box.append(table); }
